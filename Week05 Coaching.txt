import bcrypt from 'bcrypt';
import { createUser } from '../models/users.js';

export const processUserRegistrationForm = async (req, res) => {
    const { name, email, password } = req.body;

    try {
        // Generate a cryptographically secure salt with a cost factor of 10
        const salt = await bcrypt.genSalt(10);
        
        // Hash the plaintext password using the generated salt
        const passwordHash = await bcrypt.hash(password, salt);

        // Store the name, email, and securely hashed password in the database
        await createUser(name, email, passwordHash);

        if (req.flash) req.flash('success', 'Registration successful! Please log in.');
        res.redirect('/login');
    } catch (error) {
        console.error('Error registering user:', error);
        if (req.flash) req.flash('error', 'An error occurred during registration. Please try again.');
        res.redirect('/register');
    }
};

Looking at your code, you generate the salt separately using await bcrypt.genSalt(10) and then pass it to bcrypt.hash(). What went into your decision to explicitly generate and separate the salt step rather than passing the cost factor number directly into the bcrypt.hash() function?

[Sampson]: Separating the salt generation into an explicit step using bcrypt.genSalt(10) instead of passing the cost factor directly into bcrypt.hash() provides two distinct advantages:
Explicit Control and Code Clarity: Generating the salt as an independent step explicitly separates the configuration of the algorithm's work factor from the hashing operation itself. This improves code readability by making it clear that a unique, cryptographically secure salt is being generated before the computation begins.
Asynchronous Flow Predictability: Using await bcrypt.genSalt(10) ensures that the asynchronous generation of entropy completes predictably before the hashing function executes. This modular approach makes debugging and testing easier, as the salt generation and the hashing computation can be isolated and verified independently.

import { authenticateUser } from '../models/users.js';

export const processLoginForm = async (req, res) => {
    const { email, password } = req.body;

    try {
        // Verify credentials against database records
        const user = await authenticateUser(email, password);
        
        if (user) {
            // Store authenticated user profile data directly into the session payload
            req.session.user = user;
            
            if (req.flash) req.flash('success', 'Login successful!');

            if (process.env.NODE_ENV === 'development') {
                console.log('User logged in:', user);
            }

            // Redirect the authorized session to the secure dashboard view
            res.redirect('/dashboard');
        } else {
            if (req.flash) req.flash('error', 'Invalid email or password.');
            res.redirect('/login');
        }
    } catch (error) {
        console.error('Error during login:', error);
        if (req.flash) req.flash('error', 'An error occurred during login. Please try again.');
        res.redirect('/login');
    }
};

[Coach]: Great code snippet, Sampson. Attaching the user object to req.session.user makes it incredibly easy to track state across subsequent requests. Looking at your code, you store the entire user object in the session payload (req.session.user = user). What are the security or architectural tradeoffs you considered when deciding to store the entire user object in the session memory versus only storing a minimal identifier, like just the user.id?

[Sampson]: Storing the entire user object in the session memory versus storing only a minimal identifier involves clear trade-offs between architectural performance and security risks.

Architectural Trade-offs
Read Performance Boost (No Data Fetching): Storing the full object eliminates the need to query the database on every subsequent request to find out the name, email, or role tier of the logged-in user. The profile data is instantly accessible from the session storage mechanism.
Stale Data Risks: If a user's account details change (such as an administrative promotion from a standard user to an admin), the data saved inside the active session payload becomes stale. The user would retain their old permissions until they log out and log back in, or until an explicit session-update routine syncs the data. Storing only a unique identifier avoids this issue entirely, as the application is forced to fetch the freshest database records on each lifecycle loop.

Security Trade-offs
Memory Footprint and Resource Depletion: Storing full objects consumes significantly more RAM or storage space in the session store (e.g., Redis or system memory) than storing a single string or integer identifier. If the system scales up to thousands of concurrent active connections, this data load can degrade server performance.
Information Leakage Vulnerability: If the model function accidentally pulls sensitive fields, such as passwords, secret answers, or internal account flags and stores them inside the session data payload, a compromised cookie or session store breach exposes that data immediately. Minimizing the session data to just a user identifier minimizes the structural impact of a session hijack event.

[Coach]: Excellent analysis, Sampson. You perfectly balanced the read-performance benefits against the severe risks of stale data and increased memory consumption. /**
 * Middleware to check if a user is logged in
 */
export const requireLogin = (req, res, next) => {
    if (!req.session || !req.session.user) {
        if (req.flash) req.flash('error', 'You must log in to view that page.');
        return res.redirect('/login');
    }
    next();
};

/**
 * Middleware to restrict access to specific roles (like admin)
 */
export const requireRole = (allowedRole) => {
    return (req, res, next) => {
        if (!req.session || !req.session.user || req.session.user.role !== allowedRole) {
            if (req.flash) req.flash('error', 'Access denied. Authorized users only.');
            return res.redirect('/dashboard');
        }
        next();
    };
};

[Coach]: This is an excellent implementation, Sampson. The requireLogin middleware cleanly intercepts unauthenticated requests, and your requireRole function factory beautifully overcomes Express's lack of built-in parameter support for middleware. Looking closely at requireRole, you check if the user is logged in, and then check req.session.user.role !== allowedRole. If a route needs both authentication and a specific role check, what would happen if a developer accidentally put requireRole('admin') before requireLogin in their route definition, and how does your inner implementation handle or protect against that mistake?

[Sampson]: If requireRole is placed before requireLogin on an unauthenticated request:
The Risk: The server would normally crash with a TypeError because it tries to read the role of an undefined user.
The Protection: The inner implementation uses short-circuit evaluation (!req.session || !req.session.user || ...). Because the code checks for the existence of the session and user objects first, the evaluation safely stops the moment it sees no one is logged in. It completely skips reading the .role property, avoids a system crash, and safely redirects the user to the dashboard.

[Coach]: Spot on, Sampson! Your use of short-circuit evaluation prevents a fatal runtime TypeError (Cannot read properties of undefined) if the order gets mixed up.
